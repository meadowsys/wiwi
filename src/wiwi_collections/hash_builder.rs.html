68macro_rules! forward_hasher_to_inner {
69	(
70		$(
71			$(#[$meta:meta])*
72			fn $fn:ident
73			(&mut self $($params:tt)*)
74			($($args:tt)*);
75		)*
76	) => {
77		$(
78			$(#[$meta])*
79			fn $fn(&mut self $($params)*) {
80				self.inner.$fn($($args)*)
81			}
82		)*
83	}
