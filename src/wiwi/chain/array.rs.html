12impl<T, const N: usize> ArrayChain<T, N> {
13	#[inline]
14	pub fn new_uninit() -> ArrayChain<MaybeUninit<T>, N> {
15		// SAFETY: `MaybeUninit` has no initialisation requirement, so
16		// `[MaybeUninit<T>; N]` is always valid
17		unsafe { ArrayChain::from_inner(MaybeUninit::uninit().assume_init()) }
18	}
19
20	#[inline]
21	pub fn new_zeroed() -> ArrayChain<MaybeUninit<T>, N> {
22		// SAFETY: `MaybeUninit` has no initialisation requirement, so
23		// `[MaybeUninit<T>; N]` is always valid
24		unsafe { ArrayChain::from_inner(MaybeUninit::zeroed().assume_init()) }
25	}

28impl<T, const N: usize> ArrayChain<MaybeUninit<T>, N> {
29	/// Assumes all slots inside the array are initialised according to `T`'s
30	/// requirements, and converts into an array of T
31	///
32	/// Note: this implementation is currently subpar, as it does fully copy `self`
33	/// into a new container using `ptr::read`. We have to do this because, at the
34	/// time of writing:
35	///
36	/// - `transmute` is a bit too dumb, and is not able to prove `[T; N]` and
37	///   `[MaybeUninit<T>; N]` are guaranteed to be equal sized, even though
38	///   we can see and prove it
39	/// - `transmute_unchecked` is like `transmute` but without that compile time
40	///   size check, but it is unstable, and according to a code comment will
41	///   almost certainly never be stabilised (reasoning is that it's too unsafe,
42	///   too much power to give users :p, and to hopefully find other methods for
43	///   doing things without it so that it isn't needed)
44	/// - `MaybeUninit::array_assume_init` is unstable (it internally makes use of
45	///   `transmute_unchecked`)
46	///
47	/// We don't know of any other option than to perform a ptr cast,
48	/// then read from it.
49	///
50	/// # Safety
51	///
52	/// All slots in `self` must be fully initialised with valid values of `T`.
53	#[inline]
54	pub unsafe fn assume_init(self) -> ArrayChain<T, N> {
55		#[expect(clippy::as_conversions, reason = "ptr cast")]
56		let ptr = self.as_inner() as *const [MaybeUninit<T>; N] as *const [T; N];
57
58		// SAFETY: `ptr` is obtained from `self`, performed valid cast from array
59		// of MaybeUninit to array of T, and caller promises that all slots in the
60		// array are properly initialised values of `T`. Also see doc comment on
61		// this function for why this is a ptr read rather than a transmute
62		unsafe { ptr.read().into_chain() }
63	}
64}
