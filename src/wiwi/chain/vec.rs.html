12/// Creates a [`VecChain`] containing the arguments
13///
14/// Usage is same as [`vec!`], except it returns [`VecChain`] instead of [`Vec`].
15///
16/// # Examples
17///
18/// ```
19/// # use wiwi::chain::{ VecChain, ChainInner, vec_chain };
20/// let chain = vec![0u8; 32].into_chain();
21/// let chain = vec_chain![0u8; 32];
22/// ```
23#[macro_export]
24macro_rules! vec_chain {
25	[$($tt:tt)*] => { $crate::chain::VecChain::from(vec![$($tt)*]) }
26}
27pub use vec_chain;
28
29impl<T> VecChain<T> {
30	/// Creates a new vector chain without allocating any capacity
31	///
32	/// It will not allocate until it needs to, either by pushing an element,
33	/// calling the [`reserve`](Self::reserve) function to explicitly request
34	/// allocation, or something else.
35	///
36	/// # Examples
37	///
38	/// ```
39	/// # use wiwi::chain::VecChain;
40	/// // a chain thingie! yay!...
41	/// let chain = VecChain::new();
42	/// # let chain: VecChain<String> = chain;
43	/// ```
44	#[inline]
45	pub fn new() -> Self {
46		Vec::new().into_chain()
47	}
48
49	/// # Safety
50	///
51	/// You must uphold all safety requirements that [`Vec::from_raw_parts`] has.
52	#[inline]
53	pub unsafe fn from_raw_parts(ptr: *mut T, length: usize, capacity: usize) -> Self {
54		// SAFETY: caller promises to uphold safety requirements of `Vec::from_raw_parts`
55		let vec = unsafe { Vec::from_raw_parts(ptr, length, capacity) };
56		vec.into_chain()
57	}
58
59	/// Creates a new vec chain, and preallocate some memory
60	///
61	/// The amount of memory allocated will be _at least_ enough to hold `capacity`
62	/// elements without reallocating. No allocation will happen if the provided
63	/// capacity is zero, or if `T` is a ZST.
64	///
65	/// There is NO GUARANTEE that this function will allocate an exact amount
66	/// of memory, so do not rely on this for soundness. If knowing the actual
67	/// allocated capacity is important, always do so using the
68	/// [`capacity`](Self::capacity) function.
69	///
70	/// If the element type (ie. `T`) is a ZST, the vec chain will never
71	/// allocate, and will always have a capacity of `usize::MAX` bytes.
72	///
73	/// # Panics
74	///
75	/// Panics if the new capacity exceeds `isize::MAX` _bytes_ (not elements,
76	/// bytes). This is the same behaviour of [`Vec::with_capacity`].
77	///
78	/// # Examples
79	///
80	/// ```
81	/// # use wiwi::chain::VecChain;
82	/// # let mut len = 0;
83	/// # let mut initial_capacity = 0;
84	/// # let mut capacity = 0;
85	/// let chain = VecChain::new_with_capacity(10)
86	///    // chaining methods to get the len and capacity of the vec chain
87	///    .len(&mut len)
88	///    .capacity(&mut initial_capacity);
89	///
90	/// // The vector chain contains zero elements, and at least room for 10 more
91	/// assert_eq!(len, 0);
92	/// assert!(initial_capacity >= 10);
93	///
94	/// // These are all done without reallocating
95	/// let chain = (0..10)
96	///    .fold(chain, |chain, i| chain.push(i))
97	///    .len(&mut len)
98	///    .capacity(&mut capacity);
99	///
100	/// assert_eq!(len, 10);
101	/// assert_eq!(capacity, initial_capacity);
102	///
103	/// // Now however, pushing another element can make the vector reallocate
104	/// let chain = chain
105	///    .push(11)
106	///    .len(&mut len)
107	///    .capacity(&mut capacity);
108	///
109	/// assert_eq!(len, 11);
110	/// assert!(capacity >= 11);
111	///
112	/// # let mut capacity1 = 0;
113	/// # let mut capacity2 = 0;
114	/// // ZSTs never allocate and always have a capacity of `usize::MAX`
115	/// let chain1 = VecChain::<()>::new()
116	///    .capacity(&mut capacity1);
117	/// let chain2 = VecChain::<()>::new_with_capacity(10)
118	///    .capacity(&mut capacity2);
119	///
120	/// assert_eq!(capacity1, usize::MAX);
121	/// assert_eq!(capacity2, usize::MAX);
122	/// ```
123	#[inline]
124	pub fn new_with_capacity(capacity: usize) -> Self {
125		Vec::with_capacity(capacity).into_chain()
126	}
127}
