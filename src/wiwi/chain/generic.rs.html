4/// Generic (works with all types) chaining wrapper
5///
6/// This wrapper's API is very very simple, consisting of just 4 functions:
7///
8/// - [`new`](GenericChain::new) creates a new generic chainer (obviously :p)
9/// - [`map`](GenericChain::map) takes a closure that gets ownership of the inner
10///   value, and returns a new value, possibly of a different type
11/// - [`with_inner`](GenericChain::with_inner) takes a closure that can operate
12///   on a passed-in mutable reference to the inner value
13/// - [`into_inner`](GenericChain::into_inner) will unwrap the struct and return
14///   the (possibly modified) contained value.
15///
16/// The idea is to provide a generic type that can give very basic and generic
17/// chaining abilities, useful for types that don't have their own dedicated
18/// chainer.
19///
20/// This type also does not implement any traits, not even the chain ones
21/// provided by this module. The API really is just those 4 functions.
22///
23/// # Examples
24///
25/// Let's pretend [`VecChain`](super::VecChain) doesn't exist just for a moment...
26///
27/// ```
28/// # use wiwi::chain::GenericChain;
29/// // create the chain ...
30/// let numbers = GenericChain::new(Vec::<i32>::new())
31///    // ... chain push some numbers ...
32///    .with_inner(|v| v.push(1))
33///    .with_inner(|v| v.push(2))
34///    .with_inner(|v| v.push(3))
35///    .with_inner(|v| v.push(4))
36///    .with_inner(|v| v.push(5))
37///    // ... get back the inner vec, now with the pushed elements
38///    .into_inner();
39///
40/// assert_eq!(&*numbers, &[1, 2, 3, 4, 5]);
41/// ```
42#[must_use = "a chain always takes ownership of itself, performs the operation, then returns itself again"]
43#[repr(transparent)]
44pub struct GenericChain<T> {
45	inner: T
46}
47
48impl<T> GenericChain<T> {
49	#[inline]
50	pub fn new(value: T) -> Self {
51		Self { inner: value }
52	}
53
54	#[inline]
55	pub fn map<T2>(self, f: impl FnOnce(T) -> T2) -> GenericChain<T2> {
56		GenericChain::new(f(self.into_inner()))
57	}
58
59	#[inline]
60	pub fn with_inner<F, Void>(mut self, f: F) -> Self
61	where
62		F: FnOnce(&mut T) -> Void
63	{
64		void!(f(&mut self.inner));
65		self
66	}
67
68	#[inline]
69	pub fn into_inner(self) -> T {
70		self.inner
71	}
72}
73
74/// Chaining API to convert any ([`Sized`]) type to [`GenericChain<T>`] (`value.into_generic_chain()`)
75pub trait GenericChainConversion: Sized {
76	#[inline]
77	fn into_generic_chain(self) -> GenericChain<Self> {
78		GenericChain::new(self)
79	}
80}
81
82impl<T> GenericChainConversion for T {}
