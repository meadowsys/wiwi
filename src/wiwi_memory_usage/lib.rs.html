32	() => {
33		/// Types that know their own memory usage
34		#[diagnostic::on_unimplemented(
35			message = "Type doesn't know its memory usage"
36		)]
37		pub trait MemoryUsage: MemoryCompaction {
38			/// Get memory usage (including direct, indirect, and compactible memory)
39			fn memory_usage(&self) -> usize;
40		}
41
42		/// Types that always have the same memory usage, regardless of value
43		/// (notably different than [`Sized`])
44		#[diagnostic::on_unimplemented(
45			message = "Type doesn't know its static memory usage"
46		)]
47		pub trait MemoryUsageStatic: MemoryUsage {
48			const MEMORY_USAGE: usize;
49		}
50	};
51
52	(const) => {
53		/// Types that know their own memory usage

58			/// Get memory usage (including direct, indirect, and compactible memory)
59			fn memory_usage(&self) -> usize;
60		}

79pub trait MemoryCompaction {
80	/// Get expected memory usage saved after compaction
81	///
82	/// Compaction means to remove unnecessary memory usage. This would include,
83	/// for example, removing the unused capacity of a [`Vec`], by calling
84	/// [`shrink_to_fit()`](Vec::shrink_to_fit).
85	///
86	/// Types like [`Vec`] that support memory compaction _should_ reimplement
87	/// this function.
88	#[inline]
89	fn memory_usage_compacted(&self) -> usize {
90		// default is delegate to ref version of this method, so if someone
91		// implements ref, they get this one for free if they want it
92		// otherwise, no-op
93		self.memory_usage_compacted_ref()
94	}
95
96	/// Perform memory compaction, saving as much memory as possible,
97	/// returning the amount of bytes saved
98	#[inline]
99	fn compact_memory(&mut self) -> usize {
100		// default is delegate to ref version of this method, so if someone
101		// implements ref, they get this one for free if they want it
102		// otherwise, no-op
103		self.compact_memory_ref()
104	}
105
106	/// Get expected memory usage saved after compaction behind a shared reference
107	///
108	/// This is useful for types with interior mutability. Types that have or
109	/// could have (ie. with generics) should implement this method and
110	/// [`compact_memory_ref()`](MemoryUsage::compact_memory_ref).
111	/// For types that don't, the default implementation is a no-op.
112	///
113	/// See [`MemoryUsage::memory_usage_compacted`] for more details.
114	#[inline]
115	fn memory_usage_compacted_ref(&self) -> usize {
116		// default is no-op, and no bytes saved
117		0
118	}
119
120	/// Perform memory compaction behind a shared reference, saving as much
121	/// memory as possible, returning the amount of bytes saved
122	///
123	/// This is useful for types with interior mutability. Types that have or
124	/// could have (ie. with generics) should implement this method and
125	/// [`memory_usage_compacted_ref()`](MemoryUsage::memory_usage_compacted_ref).
126	/// For types that don't, the default implementation is a no-op.
127	///
128	/// See [`MemoryUsage::compact_memory`] for more details.
129	#[inline]
130	fn compact_memory_ref(&self) -> usize {
131		// default is no-op, and no bytes saved
132		0
133	}
134}
