6#[cfg(feature = "std")]
7impl_memory_usage! {
8	{ T: MemoryUsage }
9	{ T: [const] MemoryUsage }
10	// todo this is only dyn because shrink is not const?
11	dyn Vec<T>
12	|self| {
13		let vec = size_of::<Vec<T>>();
14		let elements = self.as_slice().memory_usage();
15		let remaining_capacity = (self.capacity() - self.len()) * size_of::<T>();
16		vec + elements + remaining_capacity
17	};
18	compacted |self| {
19		let vec = (self.capacity() - self.len()) * size_of::<T>();
20		let elements = self.as_slice().memory_usage_compacted();
21		vec + elements
22	};
23	compact |self| {
24		let old_cap = self.capacity();
25		self.shrink_to_fit();
26		let vec = (old_cap - self.len()) * size_of::<T>();
27
28		let elements = self.iter_mut().map(T::compact_memory).sum::<usize>();
29
30		vec + elements
31	};
32	// can compact elements by ref, but cannot shrink vec
33	ref compacted |self| self.as_slice().memory_usage_compacted_ref();
34	ref compact |self| self.iter().map(T::compact_memory_ref).sum();
35}

46	fn vec_usage() {
47		let vec_size = <*const u8>::MEMORY_USAGE + usize::MEMORY_USAGE + usize::MEMORY_USAGE;
48		let str_size = usize::MEMORY_USAGE + <*const ()>::MEMORY_USAGE;
49
50		let mut vec = Vec::with_capacity(247);
51		let vec_capacity = vec.capacity();
52		vec.extend_from_slice(&[0u8; 32]);
53		let expected = vec_size + vec_capacity;
54		let expected_compacted = vec_size + 32;
55		let expected_compact = expected - expected_compacted;
56
57		assert_eq!(vec.memory_usage(), expected);
58		assert_eq!(vec.memory_usage_compacted(), expected_compact);
59		assert_eq!(vec.compact_memory(), expected_compact);
60
61		let mut vec = Vec::with_capacity(247);
62		let vec_capacity = vec.capacity();
63		vec.extend_from_slice(&["1", "12", "123"]);
64		let expected = vec_size + (vec_capacity * str_size) + 1 + 2 + 3;
65		let expected_compacted = vec_size + (3 * str_size) + 1 + 2 + 3;
66		let expected_compact = expected - expected_compacted;
67
68		assert_eq!(vec.memory_usage(), expected);
69		assert_eq!(vec.memory_usage_compacted(), expected_compact);
70		assert_eq!(vec.compact_memory(), expected_compact);

75	fn vec_string_recursive_usage() {
76		let vec_size = <*const u8>::MEMORY_USAGE + usize::MEMORY_USAGE + usize::MEMORY_USAGE;
77		let string_size = vec_size;
78
79		let mut s1 = String::with_capacity(247);
80		let s1_capacity = s1.capacity();
81		s1.push_str("awawawa");
82
83		let mut s2 = String::with_capacity(247);
84		let s2_capacity = s2.capacity();
85		s2.push_str("12345678901234567890");
86
87		let mut s3 = String::with_capacity(247);
88		let s3_capacity = s3.capacity();
89		s3.push_str("this is a 34 character string lmao");
90
91		let mut vec = Vec::with_capacity(247);
92		let vec_capacity = vec.capacity();
93		vec.push(s1);
94		vec.push(s2);
95		vec.push(s3);
96
97		let memory_usage =
98			// vec itself
99			vec_size
100			// vec capacity
101			+ (vec_capacity * string_size)
102			// each of the 3 strings' capacities
103			+ s1_capacity
104			+ s2_capacity
105			+ s3_capacity;
106		assert_eq!(vec.memory_usage(), memory_usage);
107
108		let memory_usage_compacted =
109			// vec itself
110			vec_size
111			// vec length
112			+ (3 * string_size)
113			// each of the 3 strings' lengths
114			+ 7
115			+ 20
116			+ 34;
117
118		// expect this many bytes to be saved running compact_memory()
119		let compact_memory = memory_usage - memory_usage_compacted;
120		assert_eq!(vec.memory_usage_compacted(), compact_memory);
121		assert_eq!(vec.compact_memory(), compact_memory);
